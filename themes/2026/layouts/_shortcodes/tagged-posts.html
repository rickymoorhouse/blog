{{- /* Lists blog posts carrying a given tag, most recent first. Usage: {{< tagged-posts "api" >}} every post with the tag {{< tagged-posts "api" 5 >}} the 5 most recent Tag matching is case-insensitive, so "api" also finds posts tagged "API". */ -}} {{- $tag := .Get 0 -}} {{- $limit := .Get 1 -}} {{- $wanted := lower $tag -}} {{- $source := where site.RegularPages "Section" "in" "blog" -}} {{- $pages := slice -}} {{- range $source -}} {{- $tags := slice -}} {{- range (default slice .Params.tags) -}} {{- $tags = $tags | append (lower .) -}} {{- end -}} {{- if in $tags $wanted -}} {{- $pages = $pages | append . -}} {{- end -}} {{- end -}} {{- with $limit -}} {{- $pages = first (int .) $pages -}} {{- end -}} {{- if $pages -}}
    {{ range $pages }} {{ .Render "list-item" }} {{ end }}
{{- else -}}

No blog posts tagged {{ $tag }} yet.

{{- end -}}
